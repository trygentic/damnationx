cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(DiabloHudCore LANGUAGES CXX)
endif()
add_library(diablo_hud_core INTERFACE)
add_library(diablo_hud::core ALIAS diablo_hud_core)
target_include_directories(diablo_hud_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(diablo_hud_core INTERFACE cxx_std_17)
set(_diablo_hud_test_default OFF)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    set(_diablo_hud_test_default ON)
endif()
option(DIABLO_HUD_BUILD_TESTS "Build standalone portable HUD tests" ${_diablo_hud_test_default})
if(DIABLO_HUD_BUILD_TESTS)
    enable_testing()
    add_executable(diablo_hud_core_tests tests/core-tests.cpp)
    target_link_libraries(diablo_hud_core_tests PRIVATE diablo_hud::core)
    set_target_properties(diablo_hud_core_tests PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
    add_test(NAME diablo_hud_core COMMAND diablo_hud_core_tests)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/adapters/tests/ui_context_bridge_tests.cpp")
        add_executable(diablo_hud_ui_context_tests adapters/tests/ui_context_bridge_tests.cpp)
        target_link_libraries(diablo_hud_ui_context_tests PRIVATE diablo_hud::core)
        set_target_properties(diablo_hud_ui_context_tests PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
        add_test(NAME diablo_hud_ui_context COMMAND diablo_hud_ui_context_tests)
    endif()
    add_executable(diablo_hud_headless examples/headless.cpp)
    target_link_libraries(diablo_hud_headless PRIVATE diablo_hud::core)
endif()
